  \ifdim \skillIntensity pt < 0pt   \def\skillIntensity{0}\fi
  %
  % Draw the tag using separate spacing parameters.
  \tikz[baseline=#4]{%
    \node[draw=black!\skillIntensity!white,
          fill=white,
          % Use the passed corner style
          #9,
          inner xsep=#5,
          inner ysep=#6,
          text height=#7,
          text depth=#8,
          font=#3,
          text=black!\skillIntensity!white]
          {#1};%
  }%
